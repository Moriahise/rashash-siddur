וע"י רֵעוּתָא דְּלִבָּא שֶׁל הַמִּצְוָה הַהִיא יְתֻוֽקְּנוּ כָּל בְּחֵי' כֶּתֶר וְכָל צַלְמֵי לְבוּשֵׁי קוֹץ דְּיוּ"ד דהוי"ה שֶׁל כָּל כְּלָלוּת אבי"ע וְשֶׁל כָּל פְּרָטֵי אבי"ע דְּכָל פַּרְצוּף וּסְפִירָה דִּפְרָטֵי אבי"ע וְיִתְיַיֽחֲדוּ ד' אוֹתִיּוֹת הויה הַזּוֹ יהוה נרנח"י דִּכְלָלוּת וּפְרָטוֹת אבי"ע בִּיחוּדָא שְׁלִים וְהִיא כִּסֵּא שָׁלֵם אֲשֶׁר בָּהֶם יִתְפַּשֵּׁט שֶׁפַע הא"ס וְיִתְיַיֽחֲדוּ או"א איההיוהה בִּיחוּדָא שְׁלִים וְיִתְיַיֽחֲדוּ וְיִזְדַּוְּוֽגוּ זו"ן אידהנויה יאהדונהי זַוְּוֽגָא שְׁלִים וּמִשָּׁם יֻוֽשְׁפַּע וְיִמָּשֵׁךְ שֶׁפַע רַב וּבְרָכָה רַבָּה בְּכָל הָעוֹלָמוֹת לְזַכֵּךְ נַפְשֵׁיֽנוּ רוּחֵיֽנוּ וְנִשְׁמוֹתֵינוּ שֶׁיִּהְיוּ רְאוּיִים לְעוֹרֵר מ"ן וְאַל יְעַכֵּב שׁוּם חֵטְא וְעָוֹן וְהִרְהוּר רַע בְּקִיּוּמֵי הַתַּרְיָ"ג מִצְוֹת הֵם וְעַנְפֵיהֶם מִדְּאוֹרַיְיֽתָּא וּמִדְּרַבָּנָן ומד"ק ומד"ס וְאוֹתָם הַמִּצְוֹת אֲשֶׁר אֵין בָּנוּ כֹּחַ לְקַיְּיֽמָם וּלְבָרְרָם אַתָּה האל עוֹשֶׂה חֶסֶד לַאֲלָפִים תְּצָרֵף מַחְשְׁבוֹתֵינוּ הַטּוֹבָה כְּאִלּוּ בָּאוּ לִידֵי וְקִיַּיֽמְתִּים וְתַעֲלֶה לְרֵיחַ נִיחוֹחַ לְפָנֶיךָ וַיְהִי נֹוֽעַם אלהינו עָלֵינוּ וְכוּ':
ליקבה"ו בִּדְחִילוּ וּרְחִימוּ וּרְחִימוּ וּדְחִילוּ לְיַחֲדָא יה בֹּוֽהַּ בְּיִחוּדָא שְׁלִים בְּשֵׁם כָּל יִשְׂרָאֵל וּבְשֵׁם בִּנְיָנוֹ וּבְנוֹתֵינוּ וְאַנְשֵׁי בֵּיתֵיֽנוּ וְכָל הַנִּלְוִים אֵלֵינוּ וְהַמְסַיְּיֽעִים אוֹתָנוּ בַּעֲבוֹדַת ה'. וְהַמְהַנִּים אוֹתָנוּ מִנִּכְסֵיהֶם וּמַחֲזִיקִים בְּיָדֵינוּ לָשֶׁבֶת עַל אַדְמַת הַקַּו' יָרוּשׁ' תוב"ב וְהַמְצַוִּים אוֹתָנוּ לִלְמֹוֽד וּלְהִתְפַּלֵּל בַּעֲדָם (הֵן בִּפְרָט בְּשֵׁם פב"פ ופב"פ) וּבְשֵׁם כָּל הנרנח"י שֶׁלָּנוּ הַמִּתְיַיֽחֲסִים אל שָׁוֽרְשֵׁי נרנח"י וּמַלְבּוּשֵׁיהֶם וְהַקְּרוֹבִים לָהֶם שֶׁמִּכְלָלוֹת אבי"ע וּמִכָּל פְּרָטֵי אבי"ע דְּכָל פַּרְצוּפֵי וְסָפֵי' דִּפְרָטֵי אבי"ע:
הִנְנוּ מוּכָנִים וּמְזֻוֽמָּנִים לְקַבֵּל עָלֵינוּ עֹוֽל מ"ש תְּחִיֽלָּה. וְאַחַ"כּ עֹוֽל מִצְוֹתָיו. ומ"ע לֶאֱהֹוֽב אֶת ה' כמ"ש וְאָהַבְתָּ אֵת ה' אלהיך וְכוּ'. וְלֶאֱהֹוֽב אֶת כָּל יִשְׂרָאֵל כְּנַפְשִׁי כמ"ש וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ. ומל"ת לֹא תִשְׂנָא אֶת אָחִיךָ וְכוּ' וְלֹא תָּקוּם וְלֹא תִּטֹּוֽר וְכוּ'. וּלְהַאֲמִין בהקב"ה כמ"ש אָנֹכִי ה' אלהיך וְכוּ' וְשֶׁלֹּא לְהַאֲמִין בְּזוּלָתוֹ כמ"ש לֹא יִהְיֶה לְךָ אלהים אֲחֵרִים וְכוּ'. ומ"ע לְיַחֵד אֶת ה' כמ"ש שְׁמַע יִשְׂרָאֵל וְכוּ'. ומל"ת שֶׁלֹּא לָתוּר אַחֲרֵי הַלֵּב וְהָעֵינַיִם כמ"ש וְלֹא תָתֻוֽרוּ אַחֲרֵי לְבַבְכֶם וְאַחֲרֵי עֵינֵיכֶם וְכוּ'. וכמ"ש אל תִּפְנוּ אל הָאֱלִילִים. ומ"ע שֶׁל הַתְּשׁוּבָה כמ"ש וְשַׁבְתָּ עַד ה' אלהיך וְשָׁמַעְתָּ בְּקֹוֽלוֹ. ומ"ע שֶׁל הַוִּיֽדּוּי כמ"ש אִישׁ או אִשָּׁה כִּי יַעֲשׂוּ מִכָּל חַטֹּאת הָאָדָם לִמְעֹוֽל מַעַל בַּה' וְאָשְׁמָה הַנֶּפֶשׁ הַהִיא וְהִתְוַדּוּ אֶת חַטָּאתָם אֲשֶׁר עָשׂוּ. ומ"ע לָקוּם מִפְּנֵי הַזָּקָן וּלְהַדֵּר פְּנֵי הת"ח כמ"ש מִפְּנֵי שֵׂיבָה תָּקוּם וְהָדַרְתָּ פְּנֵי זָקֵן. ומ"ע שֶׁל הַצְּדָקָה בְּלִי נֶדֶר. וְשֶׁלֹּא לַעֲבֹוֽר עַל ל"ת שֶׁל הַצְּדָקָה בְּלִי נֶדֶר. כמ"ש וְחֵי אָחִיךָ עִמָּךְ וְכוּ' הַעֲבֵט תַּעֲבִיטֶנּוּ דֵּי מַחְסֹוֽרוֹ אֲשֶׁר יֶחְסַר לוֹ. וכמ"ש פָתֹוֽחַ תִּפְתַּח אֶת יָדְךָ לוֹ וכמ"ש לֹא תְאַמֵּץ אֶת לְבָבְךָ וְלֹא תִקְפֹּוֽץ אֶת יָדְךָ מֵאָחִיךָ הָאֶבְיוֹן. וכמ"ש נָתוֹן תִּתֵּן לוֹ וְלֹא יֵרַע לְבָבְךָ בְּתִתְּךָ לוֹ וְכוּ'. ומ"ע לְהַלְווֹת לֶעָנִי כמ"ש אִם כֶּסֶף תַּלְוֶה אֶת עַמִּי אֶת הֶעָנִי עִמָּךְ וְכוּ'. וּלְקַיֵּיֽם ז' מִצְוַת עֲשֵׂה זְכִירוֹת שֶׁצִּוָּנוּ לִזְכֹּוֽר אוֹתָם בְּכָל יוֹם. וְהֵם יְצִיאַת מִצְרַיִם. וְהַשְׁבֵּת. וְהָעֵגֶל. וּמַעֲמַד הַר סִינַי. וְהַמָּן. וּמַעֲשֵׂה מִרְיָם. וּמְחִיַּיֽת עֲמָלֵק. ומל"ת לֹא תֹאכַל כָּל תּוֹעֵבָה וְלֹא תִּֿרְצָח וְלֹא תִּֿנְאָף וְלֹא תִּֿגְנֹוֽב וְלֹא תִּגְזֹוֽל וְלֹא תַּחְמֹוֽד וְלֹא תִּתְאַוֶּה וְלֹא תְכַחֲשׁוּ וְלֹא תְשַׁקְּרוּ וְלֹא תּוֹנוּ וְלֹא תַעֲשֹׁוֽק אֶת רֵעֲךָ וְלֹא תִשָּׁבְעוּ בִשְׁמִי לַשָּׁקֶר וְלֹא תְקַלֵּל חֵרֵשׁ וְלִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשֹׁוֽל. כָּל אַלְמָנָה וְיָתוֹם לֹא תְעַנּוּן. וְשֶׁלֹּא לְחַלֵּל אֶת ה' כמ"ש וְלֹא תְחַלְּלוּ אֶת שֵׁם קָדְשִׁי. וּלְקַדֵּשׁ אֶת ה' כמ"ש וְנִקְדַּשְׁתִּי בְּתוֹךְ בְּנֵי יִשְׂרָאֵל.
