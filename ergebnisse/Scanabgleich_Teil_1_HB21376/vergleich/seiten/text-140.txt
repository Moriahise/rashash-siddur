שִׁמְךָ בְּכָל יוֹם תָּמִיד אוֹמְרִים פַּעֲמַיִם בְּאַהֲבָה:
(אַחַ"כּ יֹאמַר ק"ש לְהַחֲזִיר הַמֹּוֽחִין שֶׁבָּאוּ לז"א בַּחֲצוֹת לַיְלָה שֶׁנִּסְתַּלְּקוּ כְּשֶׁהֵאִיר הַיּוֹם יָפֶה כְּדֵי שֶׁיִּהְיֶה מוּכָן לְקַבֵּל מוֹחִין יוֹתֵר עֶלְיוֹנִים בק"ש דְּיוֹצֵר. ואעפ"י שֶׁמִּתְפַּלֵּל תְּפִלַּת הַוָּתִיקִין אֲבָל עֲדַיִיֽן זְמַן רַב שֶׁנִּסְתַּלְּקוּ וְהָיָה אֶפְשָׁר לַחֲזֹוֽר מֵאֲלֵיהֶם בְּלֹא ק"ש זֶה דְּקָרְבָּנוֹת. מ"מ יִשְׁמְרֵנוּ שֶׁמָּא ח"ו מֵסִיר כ"כ שֶׁיּוּכַל בְּמַחְשַׁבְתּוֹ בִּלְבַד לְהַחֲזִירָם כְּמוֹ שֶׁהָיוּ וְתִיֽקּוּן הָרִאשׁוֹנִים מוּעָטִים. וּמִפְּנֵי שק"ש זֶה הוּא בַּיּוֹם נִשְׁפַּע הַחֶסֶד וְהָרַחֲמִים כֹּחוֹ גָּדוֹל מִכָּל עַצְמוֹ יוֹתֵר מִשֶּׁל עַרְבִית וּמִפְּנֵי שֶׁהוּא בַּלַּיְלָה אעפ"י שֶׁגָּרוּעַ מִמֶּנּוּ בבחי' הָרֹוֽשֶׁם הַנִּשְׁאָר שָׁם אוֹר מִבְּחֵי' הַיּוֹם שֶׁעָבַר וְהַנִּשְׁאָר פה הוּא מִבְּחֵי' לַיְלָה שֶׁעָבְרָה וּלְפִיכָךְ אעפ"י שֶׁהַנִּשְׁאָר בּוֹ עַתָּה הוּא רֹוֽשֶׁם קָטָן יוֹתֵר מִשֶּׁל עַרְבִית שֶׁאֵין בּוֹ אֶלָּא בְּחַי' פְּנִימִיִּים דְּקַטְנוּת דאימ'. מ"מ יֵשׁ בּוֹ כֹּחַ עַתָּה לְהַמְשִׁיךְ לוֹ בְּחִינַת יִשְׂרָאֵל ב' הַמַּקִּיפִים הַחֲסֵרִים לוֹ שֶׁהֵם מַקִּיף קְטַנּוֹת דְּאִיֽמָּא וְהֵם לְבוּשֵׁי הו"ק דְּגַדְלוּת שֶׁלּוֹ וּפְנִימִי קַטְנוּת דְּאַבָּא. וּלְפִי שֶׁכֹּחוֹ גָּדוֹל מִשֶּׁל עַצְמוֹ עַתָּה נִמְשָׁךְ הַטִּיֽפָּה מִמָּקוֹם יוֹתֵר עֶלְיוֹן מִשֶּׁל עַרְבִית לָכֵן יְכַוֵּיֽן בְּתֵיֽבַת אֶחָד שֶׁהַהַמְשָׁכָה הַטִּיֽפָּה הוּא מִפְּנִימִיּוֹת דא"א. וּכְלָלוּת כַּוָּוֽנַת שְׁמַע הוּא כִּי שְׁמַע הִיא מָל' הַמְקַבֶּלֶת אוֹר מִזִּיֽוּוּג הזו"ן דאו"א שֶׁנַּעֲשָׂה לְהַחֲיוֹת הָעוֹלָמוֹת וְאוֹר זֶה מְקַבֶּלֶת מו"ס דְּאִיֽמָּא וִיכוֹלָה כִּי מְקַבֵּל יִשְׂרָאֵל שֶׁהוּא ז"א מחו"ג דְּאַבָּא וְגַם מְקַבֵּל ז"א מוֹחִין דְּקָט' הַחֲסֵרִים לוֹ ועי"ז מַעֲלִים מ"ן לאו"א וְזֶה מִתְעוֹרֵר לְאַבָּא ע"י שֶׁהוּא הויה לְנִגְּבוֹת פַּר' חַבָּ"ד עִם ש"ם וְכֵן עוֹשֶׂה אִיֽמָּא שֶׁהוּא אלהי"ם עִם תְּבוּנָה כְּדֵי לְהִזְדַּוֵּוֽג זִיֽוּוּג פְּנִימִי ע"י הויה שֶׁהוּא הַדַּעַת בְּכֹחַ הֶאָרָה שֶׁנִּמְשָׁכִים מִמָּקוֹם עֶלְיוֹן וּמוֹרִידִים שֶׁפַּעַם אל אֶחָד וְהוּא ז"א וְגַם הוּא מוֹשֵׁךְ הֶאָרָה לֵב' שֶׁל אֶחָד שֶׁהוּא נוּק' וע"י מְסִירַת נַפְשׁוֹ עק"ה בְּדָלֶ"ת שֶׁל אֶחָד מְקַבְּלִים רָחֵל הָעוֹמֶדֶת בְּהֵיכַל אַהֲבָה דִּבְרִיאָה עִם לֵאָה וּמַעֲלִים עַד ז"א דְּאִי' וּמִתְחַבְּרִים יַחַד וְעוֹלִים למ"ן לאו"א ואו"א מִתְיַיֽחֲדִים בְּזִיֽוּוּג בְּחַי' חִיֽצּוֹ' שֶׁלָּהֶם וְיוֹרֶדֶת הַטִּיֽפָּה שֶׁנִּמְשֶׁכֶת מִלְמַעְלָה לַיְּסוֹד דְּאִי' וְשָׁם מִצְטַיְּיֽרִים לְצָלוּחַ הֵד' מוֹחִין דְּגָד' וְתוֹךְ פְּנִימִי דו"ק שֶׁלָּהֶם שֶׁלֵּב ה' מִתְלַבְּשִׁים בו"ק דכ"א דְּתִנּוֹק שֶׁכְּנֶגְדּוֹ וכמ"ש מוֹחָם כִּי עַתָּה חָל הק"ש עָלָה נַפְשׁוֹ מָקוֹם לְחוּד הֲנֵז' שֶׁשָּׁם וְעִם קט' אֲשֶׁר עֲדַיִיֽן פּוֹסְקִים ביו"ס בְּסוֹד חו"מ וּמִתְלַבְּשִׁים בָּהֶם מֶשֶׁךְ בְּכֶתֶר לֵאָה כמעו' כִּי דְּחֶסֶד בָּעוֹלָם כּוּ' לְקַמָּן וּמֵתוּ' אֵלּוּ זו"ן דְּאִי' מֶשֶׁךְ חו"מ לַהֵיכָל אַהֲבָה דִּבְרִיאָה לְבַד שָׁם ג"כ עַצְמוֹ כנז' ב' מַקִּיף וִינִיקָה שֶׁהֵם סוֹד זו"ן הָעוֹלִים עִם נַפְשָׁהּ עִם חו"מ זֶה לַמַּקִּיף: [?])
