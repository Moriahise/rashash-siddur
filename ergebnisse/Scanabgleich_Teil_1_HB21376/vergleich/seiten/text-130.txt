[רַשִׁ"י — רֹאשׁ]
אהיה יהוה אהיה
[רַשִׁ"י — יד]
אהיה יהוה אדני
יוד הי ויו הי, יוד הי ואו הי, יוד הא ואו הא.
(וּבְכָל פַּעַם כְּשֶׁיֹּאמַר פַּר' הָעֲקֵיֽדָה יְכַוֵּיֽן לִכְבֹּוֽשׁ וּלְהַמְתִּיק הַדִּינִים וְהַגְּבוּרוֹת שֶׁבַּיְּסוֹד דז"א וִיכַוֵּיֽן שהקב"ה יַחְזֹוֽר לַעֲקֹוֽד אֶת שָׂרֵי הָאֻוֽמּוֹת שֶׁלֹּא יְקַטְרְגוּ עָלֵינוּ.)
אלהינו [יוד הי ואו הי] ואלהי [יוד הא ואו הא] אֲבוֹתֵינוּ [יוד הי ויו הי]
זָכְרֵנוּ [יוד הי ויו הי, אלף הי יוד הי] בְּזִכְרוֹן טוֹב מִלְּפָנֶיךָ וּפָקְדֵנוּ [יוד. יוד הי. יוד הי ויו. יוד הי ויו הי.] בִּפְקֻוֽדַּת יְשׁוּעָה וְרַחֲמִים [הִשְׁפַּעְתָּ כֶּתֶר] מִשְּׁמֵי שְׁמֵי קֶדֶם [יוד. יוד הי. יוד הי ויו. יוד הי ויו הי. אלף הי יוד הי.] וְזָכַר לָנוּ יהוה (אדני) אלהינו [יאהדונהי, חג"ת] אַהֲבַת הַקַּדְמוֹנִים אַבְרָהָם [חֶסֶד] יִצְחָק [גְּבוּרָה] וְיִשְׂרָאֵל [ת"ת] עֲבָדֶיךָ אֶת הַבְּרִית וְאֶת הַחֶסֶד וְאֶת הַשְּׁבוּעָה שֶׁנִּשְׁבַּעְתָּ לְאַבְרָהָם אָבִינוּ [חֶסֶד] בְּהַר הַמּוֹרִיָּה [יְסוֹד דז"א] וְאֶת הָעֲקֵדָה [מִיֽתּוּק הַדִּין] שֶׁעָקַד אֶת יִצְחָק [שֶׁהוּא גְּבוּרוֹת] בְּנוֹ עַל גַּבֵּי הַמִּזְבֵּחַ [מַלְכוּת] כַּכָּתוּב בְּתוֹרָתְךָ:
וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה והאלהים [שֶׁהֵם הַגְּבוּרוֹת שֶׁבַּיְּסוֹד דז"א] נִסָּה [לְקַשְּׁרָם] אֶת אַבְרָהָם [שֶׁהֵם הַחֲסָדִים] וַיֹּאמֶר אֵלָיו [לְיַחֲדָם עִמּוֹ וּלְמָתְקָם עַל יָדוֹ] אַבְרָהָם [שֶׁהֵם הַחֲסָדִים] וְיֹאמַר הִנְנִי [כְּלִי גְּבוּרוֹת]: וַיֹּאמֶר קַח נָא אֶת בִּנְךָ [גְּבוּרָה שֶׁבַּחֶסֶד] אֶת יְחִידְךָ [גְּבוּרָה שבת"ת] אֲשֶׁר אַהֲבַת [גְּבוּרָה שֶׁבַּנֶּצַח] אֶת יִצְחָק [שֶׁכּוֹלְלָם ג' גְּבוּרוֹת דחג"ת] וּלְךָ [לְמַתְּקָן טוּר חֲסָדִים לְאוֹר יָשָׁר] לְךָ אל אֶרֶץ הַמֹּוֽרִיָּה [שֶׁהוּא הַיְּסוֹד דז"א לְמַתֵּק ג' גְּבוּרוֹת דתנה"י וחו"ב חוֹזֵר] וְהֶעֱלָהוּ
