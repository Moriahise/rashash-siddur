[לחח"ן בג"ה דס"ג דְּפַרְצוּפֵי מ"ה וב"ן דְּיִשְׂרָאֵל סָבָא וּתְבוּנָה דז"א דַּעֲשִׂיָּה דִּכְלָלוּת וּרְבִיעִיָּה הַתַּחְתּוֹנָה:]
אלהי נְשָׁמָה [ה' בְּחֵי' נרנח"י דִּכְלָלוּת אבי"ע] שֶׁנָּתַתָּ בִּי [נִשְׁמַת אָדָם הָרִאשׁוֹן] טְהוֹרָה [כְּלָלוּת נרנח"י דְּחַיָּה דִּכְלָלוּת] אַתָּה בְּרָאֲתָה [כְּלָלוּת נרנח"י דִּנְשָׁמָה דִּבְרִיאָה] אַתָּה יְצַרְתָּהּ [כְּלָלוּת נרנח"י דְּרוּחַ דִּיצִירָה] אַתָּה נִפְחֲתָה בִּי [כְּלָלוּת נרנח"י דְּנֶפֶשׁ דַּעֲשִׂיָּה] וְאַתָּה מְשַׁמְּרָהּ [מִן הַחִיצוֹנִים] בְּקִרְבִּי וְאַתָּה [מַלְכוּת דִּכְלָלוּת] עָתִיד לְטָלֶה מִמֶּנִּי [למ"ן] וּלְהַחֲזִירָהּ בִּי [בְּכָל בָּקָר וְגַם] לְעָתִיד לָבֹא כָּל זְמַן שֶׁהַנְּשָׁמָה בְּקִרְבִּי מוֹדֶה אֲנִי לְפָנֶיךָ יהוה (אדני) יאהדונהי אלהי ואלהי אֲבוֹתַי רִבּוֹן כָּל הַמַּעֲשִׂים אֲדוֹן כָּל הַנְּשָׁמוֹת: בָּרוּךְ אַתָּה יהוה (אדני) יאהדונהי הַמַּחֲזִיר נְשָׁמוֹת לִפְגָרִים מֵתִים:
(אַחַ"כּ יֹאמַר הח"י בְּרָכוֹת שֶׁהֵם מֵהַנּוֹתֵן לַשֶּׂכְוִי עַד סוֹף בִּרְכַּת הַתּוֹרָה לְתַקֵּן ח"י בְּחַי' וְכוּ' עַיֵּיֽן בָּס' כ"ס דַּף פ"ו דְּפוּס ירוש"ו דַּף י"ח ע"ג וְעַיֵּיֽן אוֹר יָשָׁר וְאוֹר חוֹזֵר גַּם זֶה שָׁם גַּם הַקְדָּמַת הַנּוֹתֵן לַשֶּׂכְוִי שָׁם:[?])
[גַּם יְכַוֵּיֽן בְּחַי' בְּרָכוֹת אֵלּוּ לְתַקֵּן כֵּלִים וְאוֹרוֹת דחו"י והו"ח די"ב פַּרְצוּפִים הַכְּלוּלִים בְּפַרְצוּף אֶמְצָעִי דז"א דַּעֲשִׂיָּה דִּכְלָלוּת וּרְבִיעִיָּה הַתַּחְתּוֹנָה וּלְהַמְשִׁיךְ לָהֶם מוֹחֵי' ונרנח"י דמ"ה וב"ן דחו"י והו"ח:[?]]
