ח"ק הֵיכַל ק"ק הוּא גַּנָּב שֶׁהוּא כֶּתֶר חָכְמָה וּבִינָה וּבְכֶתֶר הנז"ל עָלוּ הָנֵק' ע"י וא"א ובו"ק עָלוּ הָנֵק' או"א וְנוּק' וְנִכְנַס כַּנַ"ל שֶׁהֵם נַק' חו"ב עִי' והו"ק שנג' נַק' יססו"ת מִמַּה כִּי כַּנַ"ל עַל הנרנח"י עִם מ"ן וחו"ב עַי' וְהָנֵק' בְּקַדִּישׁ. [?]
יְכַוֵּיֽן מֶשֶׁךְ ל"ם דְּחַבַּ"ד דל"ם ומ"ב חו"ן ז' תֵּיֽבוֹת מִן הַשֶּׁבַח עַד יִתְגַּדֵּל והו"ק כו'. [?]
שָׁם ז"א וְנוּק' וישסו"ת כנז' זַ"ל בְּעֵרֶךְ הנז"ל וְהַפְּרָטִים עוֹד לְכֶתֶר חָכְמָה וּבִינָה וחג"ת כנז' וְחָל בְּכֶתֶר נִמְשָׁךְ לוֹ מו' דְּאַבָּא וּבִכְלָלוּת לְבוּשׁ מִבִּינָה עַל יִתְגַּדַּל וְיִתְקַדַּשׁ שֶׁהֵם שְׁנֵי כַּנַ"ל בְּמִלַּת רָבָא וּבִכְלָלוּת שְׁנֵי חַיּוֹת שָׁם נִמְשָׁכוֹת עַד מְשִׁיחֵיֽהּ שֶׁכֻּלָּם מֵאִיֽמָּא נִמְשָׁךְ מו' דְּאַבָּא לַחָכְמָה וּבִכְלָלוּת כֹּחַ הַיְּסוֹד שָׁם מִן יְהֵא שְׁמֵיֽהּ רַבָּא עַד בְּעָלְמָא בְּמִלַּת בְּעָלְמָא נִמְשָׁךְ מו' דְּאַבָּא לְבֵינָהּ והנה"י הֲנֵז' מַקִּיפִים לִכְחַבָּ"ד חג"ת כנז' וּבְאֻמָּן אַחֲרוֹן נִמְשָׁךְ מו' דְּאַבָּא לִכְחַבָּ"ד וּבְמִלַּת וַיַּקְדֵּם נִמְשָׁךְ מו' דְּאַבָּא לחג"ת כַּנָּז' וּבְמִלַּת רָבָא נִמְשָׁךְ מו' דְּאַבָּא לנה"י וְהֵם כַּעֲלוּתָם מִקַּטְנוּת נַק' ז"א בְּעֵרֶךְ כְּלָלוּת הֲנֵז' נִכְתְּבוּ ז"א הַגְּדוֹלִים והו"ק דז"א נַק' יעו"ר כַּנָּז' ז"א בְּעֵרֶךְ הֲנֵז' נַק' ז"א שֶׁהֵם זו"ן הַגְּדוֹלִים וּכְחַבַּ"ד ז"א שֶׁהֵם זו"ן הַגְּדוֹלִים נִמְשָׁךְ לָהֶם מו' דְּאִיֽמָּא במ"ב דָּמוֹ' דמ"ה דְּמַס' וָוִין דִּכְלָלוּת חַיּוֹת וּנְשָׁמוֹת וּמִתְחַבֵּר בַּתְּחִלָּה וְיִתְגַּדֵּל והו"ק דז"א שֶׁהֵם ע"י וְנִפְרָטִים עוֹד לְכֶתֶר חָכְמָה וּבִינָה וחג"ת כנז' יַחֵל בְּכֶתֶר וּבְחָכְמָה נִמְשָׁךְ לָהֶם מו' דְּאִיֽמָּא בְּמִלַּת דַּי.
