וּלְבָרֵר חֶלְקֵי הַנְּפָשׁוֹת שֶׁנִּפְגְּמוּ וְנָפְלוּ בְּנֹוֽגַהּ דאבי"ע. וְנֹוֽגַהּ כְּאוֹר תִּהְיֶה אל מְקוֹם הַקֹּדֶשׁ. אֲנָא ה' הַפֶּרֶד נָא מֵעָלֵינוּ צַד הָרַע שביצה"ר וְתִזְכֵנוּ וַתִּלְבָּשֵׁנוּ בְּכֹחַ היצה"ט.
וע"י מַעֲשֵׂה הַמִּצְוֹת הַלָּלוּ יְתֻוֽקְּנוּ כָּל בְּחֵי' מַלְכֻיוֹת וְכָל צַלְמֵי לְבוּשֵׁי ה' אַחֲרוֹנָה דַּהֲוָיָה נְפָשׁוֹת שֶׁל כָּל כְּלָלוּת אבי"ע וְשֶׁל כָּל פְּרָטֵי אבי"ע דְּכָל פַּרְצוּף וּסְפִירָה דִּפְרָטֵי אבי"ע:
הֲרֵי אֲנִי מַאֲמִין בָּהּ' אלהי יִשְׂרָאֵל אֲשֶׁר בָּרָא שָׁמַיִם וָאָרֶץ וְכָל צְבָאָם הוּא הָיָה וְהוּא הוה וְהוּא יִהְיֶה. הוּא הָיָה קֹוֽדֶם דִּבְרָא עָלְמָא. וְהוּא הוה אַחַר דְּאֶבְרֵי עָלְמָא. וְהוּא יִהְיֶה לעת"ל הוּא קַיָּיֽם וּשְׁמוֹ קַיָּיֽם וְכִסְאוֹ וּמַלְכוּתוֹ וֶאֱמוּנָתוֹ לָעַד קַיֶּיֽמֶת. הוּא שֹׁוֽרֶשׁ הָאֱלֹהוּת הוּא נְשָׁמָה לְכָל הַנְּשָׁמוֹת. שְׁמוֹ קַיָּיֽם עֶצֶם הָאֱלֹהוּת הַשֵּׂכֶל הַפְּנִימִי הַנּוֹבֵעַ מִמֶּנּוּ. וְכִסְאוֹ נָכוֹן וּמַלְכוּתוֹ וֶאֱמוּנָתוֹ לָעַד קַיֶּיֽמֶת. ה' שֶׁבָּאֱלֹהוּת ה' אלהי יִשְׂרָאֵל שְׁמוֹ הוּא א' וּשְׁמוֹ א' בָּרוּךְ הוּא וּבָרוּךְ שְׁמוֹ וְזִכְרוֹ לְעוֹלְמֵי עַד אָמֵן:
וַהֲרֵי אֲנִי מַאֲמִין בְּעִילָא עַל כָּל הָעִיֽלּוֹת בקבה"ו הַמִּתְלַבֵּשׁ בי"ס דְּאָצֵי' ע"י שְׁמוֹ הַגָּדוֹל וּמִתְגַּלֶּה בת"ת וּמָל' וְהַלָּלוּ אֲנִי מַעֲלֶה כַּוָּוֽנַת הַתְּפִלּוֹת וְהִבָּרְכוּ' וְהַהוֹדָאוֹת וְלִיֽמּוּד הַתּוֹרָה וְקִיּוּם הַמִּצְוֹת. וְכָל דְּבָרַי וְכָל מַעֲשֵׂי הַטּוֹבִים לְהוֹרִיד שֶׁפַע בְּרָכָה רַבָּה מֵהֶאָרָתוֹ. מא"ק לאו"א ומאו"א לזו"ן. וּלְיַחֵד קבה"ו יִחוּד גָּמוּר יאהדונהי הוּא הָיָה וְהוּא הוה וְהוּא יִהְיֶה הוּא הָיָה טְמִיר וּגְנִיז בְּכֶתֶר וְהוּא הַהוֹוֶה שֶׁמִּתְלַבֵּשׁ בי"ס דְּאִצִּי'. וְהוּא יִהְיֶה שֶׁמִּתְגַּלֶּה בת"ת וּמָל'. וְהַלָּלוּ אֲנִי מוֹסֵר נַפְשִׁי רוּחִי וְנִשְׁמָתִי עַל קְדֻוֽשַּׁת שְׁמוֹ הַגָּדוֹל. וּמְקַבֵּל עָלַי עֹוֽל מַלְכוּתוֹ כַּכָּתוּב שְׁמַע יִשְׂרָאֵל וְכוּ' בשכמל"ו. ה' הוּא האלהים וְכוּ'. ה' מֶלֶךְ וְכוּ' וְהָיָה ה' לַמֶּלֶךְ וְכוּ'. וְעַתָּה ה' אלהים הוֹשִׁיעֵנוּ בְּקָרוֹב לְמַעַן שִׁמְךָ כִּי אַתָּה אָבִינוּ. אַתָּה תּוֹחַלְתֵּנוּ. וּבְךָ בָּטַחְנוּ. אַתָּה תָקוּם תְּרַחֵם צִיּוֹן וְכוּ' ע"י עֲבָדֶיךָ הָאֵלֶּה הֵם בָּנֶיךָ הַיּוֹדְעִים אֱלֹהוּתְךָ הַמְפֻרְסָמִים אֱמוּנָתֶךָ. הַמְצַפִּים יְשׁוּעָתְךָ. גִּלָּה כְּבוֹד מַלְכוּתֵךְ עָלֵינוּ מְהֵרָה וְיָדַע כָּל פָּעוּל וְכוּ' אוֹדְךָ עַל כִּי נוֹרָאוֹת נִפְלֵאתִי נִפְלָאִים מַעֲשֶׂיךָ וְנַפְשִׁי יֹוֽדַעַת מְאֹד וְנָהָר יֹוֽצֵא מֵעֵדֶן לְהַשְׁקוֹת אֶת הַגָּן וּמִשָּׁם יִפָּרֵד וְהָיָה לְאַרְבָּעָה רָאשִׁים שֵׁם הא' פִּישׁוֹן הוּא הַסֹּוֽבֵב אֵת כָּל אֶרֶץ הַחֲוִילָה אֲשֶׁר שָׁם הַזָּהָב וְזָהָב הָאָרֶץ הַהִיא טוֹב שָׁם הַבְּדֹוֽלַח וְאֶבֶן הַשֹּׁהַם. וְשֵׁם הַנָּהָר הַב' גִּיחוֹן הוּא הַסּוֹבֵב אֵת כָּל אֶרֶץ כּוּשׁ. וְשֵׁם הַנָּהָר הַג' חִדֶּקֶל הוּא הַהֹוֽלֵךְ קִדְמַת אַשּׁוּר. וְהַנָּהָר הֵד' הוּא פְּרָת. אלהי אַתָּה וַאֲרוֹמִמְךָ אוֹדֶה שִׁמְךָ כִּי עָשִׂיתָ פֶּלֶא עֵצוֹת מֵרָחוֹק אֱמוּנָה אֹוֽמֶן:
