(וְיֹאמַר הַקָּרְבָּנוֹת לְתַקֵּן פְּנִימִיּוּת וְחִיצוֹנִיּוֹת דִּפְנִימִיּוֹת דְּעָשֵׁי' דאבי"ע בְּעֹוֽבִי לְהַעֲלוֹתָם עִם פְּנִימִיּוּת וְחִיצוֹנִיּוֹת דְּחִיצוֹנִיּוֹת הַנִּתְקָן ע"י מִצְוֹת מַעֲשִׂיּוֹת)
(וִיכַוֵּיֽן עַצְמוֹ לְהִתְפַּלֵּל בְּהַכְנָעָה וּבְקוֹל נָמוּךְ וּבְשִׂמְחָה וְלֹא בְּעַצְבוּת וּמִתּוֹךְ הַסֵּפֶר וְיִסְתַּכֵּל עֵינָיו בּוֹ. וְכָל דָּבָר שֶׁיִּסְתַּכֵּל יְכַוֵּיֽן שֶׁעֵינָיו הֵם סוֹד חו"ב. וְכָל פָּסוּק שֶׁבַּתְּפִלָּה יֹאמְרֶנּוּ בִּטְעָמִים הַרְאוּיָם לוֹ. וּכְדֵי שֶׁתָּחוּל עָלָיו קְדֻוֽשָּׁה עֶלְיוֹנָה וְתִתֵּן לוֹ כֹּחַ לִקְשֹׁוֽר הָעוֹלָמוֹת בִּתְפִלָּתוֹ יַעֲשֶׂה עַצְמוֹ כִּסֵּא לִקְדֻוֽשָּׁה. וְיַחְשֹׁוֽב כִּי רֹאשׁוֹ כִּסֵּא לַהֲוָיָה בְּנִיֽקּוּד קָמַץ. וּב' מוֹחִין שֶׁבּוֹ הֵם סוֹד י"ה שֶׁהֵם בחו"ב בְּיָמִין שֶׁבָּהֶם יְכַוֵּיֽן לַהֲוָיָה בְּנִיֽקּוּד פָּתַח וּבִשְׂמֹאל בַּהֲוָיָה בְּנִיֽקּוּד צֵירֵי. וּלְעֵינָיו יְכַוֵּיֽן כִּי עַיִ"ן הוּא מִסְפַּר ה' הֲוָיוֹת וּמִפְּנֵי שֶׁהֵמָּה בִּשְׁנַיִם יְכַוְּוֽנֵם בְּמִיֽלּוּי הַהִין. וּלְאָזְנָיו יְכַוֵּיֽן אֹזֶן שֶׁהוּא ס"ג חוּץ מֶה' אַחֲרוֹנָה שֶׁהִיא בהי"ן. ועי"ז יִשְׁמַע אֵיזוֹ קְדֻוֽשָּׁה עֶלְיוֹנָה וּבְחֹוֽטֶם יְכַוֵּיֽן שֶׁמִּסְּפָרוֹ ס"ג אוּלַי יָרִיחַ בּוֹ רֵיחַ קְדֻוֽשָּׁה. וּלְפֶה יְכַוֵּיֽן שֶׁמִּסְפָּרָהּ ס"ג וכ"ב אוֹתִיּוֹת שֶׁבְּחָמֵשׁ מוֹצָאוֹת הַיּוֹצֵאת מִמֶּנָּה אוּלַי יַשִּׂיג לְרוּחַ ה' שֶׁיְּדַבֵּר בּוֹ וְלֵב' זְרוֹעוֹת יְכַוֵּיֽן לֵב' הֲוָיוֹת בְּנִיֽקּוּד סֶגוֹל לִימִינוֹ וּבַנִּיֽקּוּד שֶׁבָּא לִשְׂמֹאלוֹ. וּלְגוּפוֹ הוּא סוֹד ו' שֶׁל שֵׁם שֶׁהוּא ז"א יְכַוֵּיֽן הויה בְּנִיֽקּוּד חוֹלָם. וְלֵב' יַרְכוֹתָיו יְכַוֵּיֽן לֵב' הֲוָיוֹת א' בְּנִיֽקּוּד חִירִיק לְיָמִין וה' בְּנִיֽקּוּד קִבּוּץ לִשְׂמֹאלוֹ. וְלִבְרִיתוֹ יְכַוֵּיֽן הויה בְּנִיֽקּוּד שׁוּרֻק וָא"ו. וְלַעֲטָרָה שֶׁלּוֹ וְהִיא סוֹד ה' אַחֲרוֹנָה שֶׁל שֵׁם שֶׁהִיא מָל' יְכַוֵּיֽן הויה בְּלִי נְקֻוֽדּוֹת. וְכָל אֵלּוּ הַכַּוָּוֽנוֹת שֶׁל הֲוָיוֹת הֵם בִּפְנִימִיּוּתוֹ אֲבָל בְּחֵיצוֹ' בְּצֶלֶם אלהים עָשָׂה אוֹתוֹ, כִּי הא' שֶׁבּוֹ הִיא גֻּולְגָּלְתָּא כֶּתֶר, וְהֵל' ג' מוֹחִין חַבָּ"ד, וְהָהּ' ה"ק מֵחֶסֶד עַד הוֹד שֶׁמִּתְפַּשְּׁטִים בָּהֶם הַמֹּוֽחִין הֲנֵז', וְהֵי' ה"ס בְּרִיָּתוֹ וּמִתְפַּשְּׁטִים בּוֹ כְּלָלוּת הָמוּ' וְהֵם' הִיא עֲטָרָה שֶׁלּוֹ שֶׁהִיא מַלְכוּת וּמִתְפַּשְּׁטִים בּוֹ כְּלָלוּת הַמֹּוֽחִין:)
