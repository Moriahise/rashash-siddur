כִּיהוּדָא וְעוֹד לְהַסְכִּים ע"ד הָאִיֽסּוּר כִּי אֲנִי יוֹדֵעַ טִרְחָתָם וְהַהוֹצָאָה הַגְּדוֹלָה שֶׁעָלָה לָהֶם וּזְכוּת הַהוֹצָאָה הָאוֹר הַנִּפְלָא לְהָאִיר עוֹלָם, נִזְכֶּה שֶׁיְּקֻוֽיַּם בַּקָּשָׁתֵנוּ אוֹר חָדָשׁ עַל צִיּוֹן תָּאִיר וְנִזְכֶּה כֻּוֽלָּנוּ מְהֵרָה לְאוֹרוֹ בִּמְהֵרָה בְּיָמֵינוּ אָמֵן.
כַּעֲתִירַת הַקָּטָן שֶׁבִּקְטַנִּים
— מְנַחֵם מֵעַנְדְּכִין הַיְיֽלְפֵּרִין.
■ הַסְכָּמַת הָרַבָּנִים הַגְּאוֹנִים בה"צ הָאַשְׁכְּנַזִּים הי"ו ■
נוֹדֶה ה' מְאֹד בְּפִינוּ וּבְתוֹךְ רַבִּים נְהַלְּלֶנּוּ שֶׁהֶחֱיָיֽנוּ וְקִיַּמְנוּ וְהִגִּיעָנוּ לַחֲזוֹת בְּנֹוֽעַם זִיו יְקָר תִּפְאֶרֶת הַסִּיֽדּוּר הַקָּדוֹשׁ מִכְּבוֹד קְדֻוֽשַּׁת מוֹרֵיֽנוּ וּמֵאוֹרְיָנוֹ הַקָּדוֹשׁ וְהַטָּהוֹר, שַׂר בֵּית הַזֹּוֽהַר הַמְקֻוֽבָּל הָאֱלֹקִי מָרָן רַב שַׂר שָׁלוֹם שַׁרְעַבִּי זי"ע אֲשֶׁר הָיָה סָפוּן וְטָמוּן, וְלֹא זָכוּ לְאוֹרוֹ אַף א' בָּעִיר וְכוּ', עַד כֹּה אֲשֶׁר הֵעִיר ה' אֶת רוּחַ הָנְהוּ גַּבְרָא יַקִּירָא הָרַבָּנִים הַנִּכְבָּדִים הַחֲסִידִים הַמֻּוֽפְלָאִים יִרְאֵי ה' וְכוּ' הָר' רְאוּבֵן הַאֲאַן וְהַר' שְׁמוּאֵל קִירְשֶׁענְבּוֹים וְהַר' יַעֲקֹב זַאנֶענְפֶּעלְד והרה"ג הַמְקֻוֽבָּל הָאֱלֹקִי וְכוּ' הָר' יוֹם טוֹב יְדִיד הַלֵּוִי וּבְנוֹ הָר' אֱלִיעֶזֶר יְדִיד הַלֵּוִי הי"ו וְנַפְשָׁם אוֹתָהּ לִהְיוֹת מִתְעַסְּקִים בַּקֳּדָשִׁים לְמַעַן זְכוּת אֶת הָרַבִּים, וּלְהָאִיר עַל הָאָרֶץ בְּאוֹר הַתּוֹרָה, וְהֶרְאוּ לָנוּ אֶת שֶׁכְּבָר נִדְפְּסוּ עַל יָדָם חֵלֶק א' בִּנְיָיֽר טוֹב וְיָפֶה, וְאוֹתִיּוֹת מַזְהִירוֹת מִשִּׂמְחַת לֵב וְעֵינַיִם מְאִירוֹת, וְעוֹד יָדָם נְטוּיָה לְבָרֵךְ עַל הַמֻּוֽגְמָר ברצ"ה. מַה מְאֹד שָׂמַח לִבֵּנוּ וְתַעֲלֹוֽזְנָה כִּלְיוֹתֵינוּ, וְהִנֵּה מִי יַעְצֹוֽר כֹּחַ לְתָאֵר וּלְבָאֵר גְּדֻוֽלַּת תְּהִלַּת הַאי סִדְרָא קַדִּישָׁא, לֹא בָּאנוּ אֵיֽפוֹא בָּזֶה רַק לִגְדֹּוֽר גָּדֵר, וְלִגְזוֹר בְּאִיֽסּוּר עַל כָּל הַמַּדְפִּיסִים שֶׁבָּעוֹלָם לְבַל יְעִיזוּן וּבַל יִשְׁכֹּוֽן לַחֲזֹוֽר לְהַדְפִּיס הַאי סִדְרָא קַדִּישָׁא, עַד כְּלוֹת חֲמִשָּׁה עָשָׂר שָׁנָה מִיּוֹם צֵאתוֹ מִבֵּית הַדְּפוּס. וְזֹאת שֵׁנִית, לִבְנוֹ לְחוֹקְקֵי יִשְׂרָאֵל, מִשְׁכוּ וּקְחוּ לָכֶם בְּרָכָה אל תּוֹךְ בָּתֵּיכֶם, קָנוּ וְזָכוּ לָכֶם הַאי סָפְרָא דתוה"ק דֵּעָה, מִיַּד הַמּוֹ"ל בִּמְחִיר אֲשֶׁר יָשִׁיתוּ עָלָיו. לְמַעַן יַעֲלֶה בְּיָדָם לִהְיוֹת מַדְפִּיסִים וּמַעֲלָה יֶתֶר הַחֲלָקִים מִסִּיֽדּוּר הַקָּדוֹשׁ, כִּי כָּל מִי שֶׁיְּסַיֵּיֽעַ לָזֶה (גַּם מִי שֶׁאֵין לוֹ יָד וְשֵׁם בְּחָכְמַת הַקַּבָּלָה) שְׂכָרוֹ הַרְבֵּה מְאוֹד וּזְכוּת הָרַבִּים יִהְיֶה תָּלוּי בּוֹ, אַשְׁרֵי אֱנוֹשׁ יַעֲשֶׂה זֹּאת וּבֶן אָדָם יַחֲזִיק בָּהּ זְכוּת תוה"ק וּזְכוּת מוֹרֵיֽנוּ הַקָּדוֹשׁ הַמְחַבֵּר זי"ע נִפְלָאָה הִיא שֶׁתָּגֵן בְּאֶלֶף הַמָּגֵן עַל כָּל הָעוֹשִׂים וְהַמַּעֲשִׂים לָזֶה, לִשְׁכּוֹן בְּאָהֳלָם, הָנֵי שָׁלֹשׁ אלה בְּמַזָּלָא עִיֽלָּאָה תַּלְיִין וְלַחֲזוֹת בְּנֹוֽעַם ה' בב"א כיר"א.
כַּעֲתִירַת הבד"צ דפעיה"ק יְרוּשָׁלִַם תובב"א. הַמַּחְכִּים לִישׁוּעָה קְרוֹבָה לָבֹא. וְחוֹתְמִים בִּבְרָכָה יוֹם כ"א לְחֹדֶשׁ מַרְחֶשְׁוָן פעיה"ק תובב"א.
