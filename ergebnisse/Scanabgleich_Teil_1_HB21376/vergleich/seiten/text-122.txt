[תְּפִיֽלִּין וּבִרְכוֹתֵיהֶם הֵם מִצְוַת עֲשֵׂה שהז"ג שֶׁהִיא בחג"ת דחג"ת וחג"ת דנה"י דחו"ב דז"א דַּאֲצִילוּת וְהַכַּוָּוֽנָה לְזִוּוּג פַּרְצוּפֵי' הָעֶלְיוֹנִים דַּאֲצִילוּת וּלְהַמְשִׁיךְ מֵהֶם צַלְמֵי הַמֹּוֽחִין לחו"ב הֲנֵז' דז"א דַּאֲצִילוּת בַּבְּרָכָה נִמְשָׁךְ הל"ם דְּצֶלֶם וע"י הַמַּעֲשֶׂה שֶׁל הַמִּצְוָה נִמְשָׁךְ הַצ' דְּצֶלֶם בִּפְנִימִיּוּת הז"א וְגַם ע"י תש"י יְכַוֵּיֽן לְהַפְרִיד הַקָּלִי' מֵפֵר' נה"י (שֶׁהֵם זו"ן ויעקו"ר) דְּחֵיצוֹ' דְּחִיצוֹנֵי' דִּבְרִיאָה וְגַם מִפְּנִימָיו' דְּחֵיצוֹ' ז"ע דַּעֲשִׂיָּה וע"י תש"ר לְהַפְרִיד הַקָּלִי' מֵחֵיצוֹ' נה"י דְּאָצֵי' וּמֵחִיֽצּוֹ' דְּחֵיצוֹ' ז"ע דִּבְרִיאָה וּמִפְּנִימִי' דְּחֵיצוֹ' ז"ע דִּיצִירָה וּלְהַמְשִׁיךְ בָּהֶם מוֹחִין דְּחִיּוֹת וְדִנְשָׁמוֹת עִם הנרנח"י שֶׁבָּהֶם כַּאֲשֶׁר יִתְבָּאֵר כ"א בִּמְקוֹמוֹ:]
*בָּרוּךְ אַתָּה יהוה (אדני) יאהדונהי אלהינו מֶלֶךְ הָעוֹלָם אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לְהַנִּיחַ תְּפִיֽלִּין:
[וִיכַוֵּיֽן לְקַיֵּיֽם מ"ע לִלְבֹּוֽשׁ תְּפִיֽלִּין וְהִיא מ"ע שהז"ג שֶׁהִיא בחג"ת דְּפַרְצוּפֵי נה"י וחג"ת דז"א. וִיקַיֵּיֽם מ"ע שֶׁנֶּאֱמַר וּקְשַׁרְתָּם לְאוֹת עַל יָדֶךָ וּלְטֹטָפֹת בֵּין עֵינֶיךָ. וע"י כַּוָּנַת הֲכָנָה זוֹ אָז יִהְיֶה כֹּחַ בְּכַוָּוֽנָתוֹ בִּבְרָכָה לְהַעֲלוֹת מ"ן וּלְהַמְשִׁיךְ הָמוּ' לזו"ן.]
[הֶעָרַת תַּחְתִּית: *) א"ה כַּוָּוֽנַת מַטְבֵּעַ בִּרְכַּת הַתְּפִלִּין יְכַוֵּיֽן כנז"ל בְּבִרְכַּת הַצִּיצִית וְאַף שֶׁנִּסְתַּפֵּק הָרַב עָלַי נָהָר אִם לְכַוֵּיֽן או לֹא וְהֶעֱלָה שָׁם בָּס' בְּדַף ח' ע"א אוֹת ל' שֶׁלֹּא לְכַוֵּיֽן כְּבָר שָׁם בְּהַגָּהַת הָר' שד"ה יל"ו תֵּרֵץ דְּבָרָיו ע"ש וְכֵן כָּתַב הָר' פֶּתַח עֵינַיִם בדע"ד ע"ד שֶׁכַּוָּוֽנַת בִּרְכַּת הַתְּפִלִּין הִיא מִלְּבַד בְּחַי' הִרְשִׁימוּ שֶׁל הַתְּפִלִּין דְּיוֹם תְּמוֹל וְכֵן ג"כ בִּישִׁיבָתֵיֽנוּ נוֹהֲגִים לְכַוֵּיֽן ויעב"א. א"א נ"י וְהַצָּעִיר אֱלִיעֶזֶר יְדִיד הַלֵּוִי.] [תִּיֽקּוּן שֶׁנִּדְפַּס בְּתַחְתִּית הָעַמּוּד:] וְגַם מֵחִיצוֹנִיּוֹת דְּחֵיצוֹ' כְּחַבָּ"ד חג"ת (שֶׁהֵם עי"ו ואו"נ ואו"א וישסו"ת) דִּיצִירָה וְגַם.
