[רַשִׁ"י — רֹאשׁ]
אהיה יהוה אהיה
[רַשִׁ"י — יד]
אהיה יהוה אדני
יוד הי ויו הי, יוד הי ואו הי, יוד הא ואו הא.
וְתִכָּנֵס לָנוּ לִפְנִים מִשּׁוּרַת הַדִּין וּבְטוּבְךָ הַגָּדוֹל [חֲסָדִים וְרַחֲמֵי דְּאִי'] יָשׁוּב חֲרוֹן אַפְּךָ מֵעִמָּךְ וּמֵעִירָךְ וּמֵאַרְצֵךְ וּמִנַּחֲלָתָךְ: וְקִים לָנוּ יהוה (אדני) אלהינו אֶת הַדָּבָר שֶׁהִבְטַחְתָּנוּ בְּתוֹרָתָךְ עַל יְדֵי מֹשֶׁה עַבְדְּךָ כָּאָמוּר וְזָכַרְתִּי אֶת בְּרִיתִי יַעֲקוֹב וְאַף אֶת בְּרִיתִי יִצְחָק וְאַף אֶת בְּרִיתִי אַבְרָהָם אֶזְכֹּר וְהָאָרֶץ אֶזְכֹּר:
וְנֶאֱמַר וְאַף גַּם זֹאת בִּהְיוֹתָם בְּאֶרֶץ אֹיְבֵיהֶם לֹא מְאַסְתִּים וְלֹא גְעַלְתִּים לְכַלֹּוֽתָם לְהָפֵר בְּרִיתִי אִתָּם כִּי אֲנִי יהוה (אדני) אלהיהם: וְנֶאֱמַר וְזָכַרְתִּי לָהֶם בְּרִית רִאשֹׁנִים אֲשֶׁר הוֹצֵאתִי אֹוֽתָם מֵאֶרֶץ מִצְרַיִם לְעֵינֵי הַגּוֹיִם לִהְיֹוֽת לָהֶם לאלהים אֲנִי יהוה (אדני): וְנֶאֱמַר וְשָׁב יהוה (אדני) אלהיך אֶת שְׁבוּתְךָ וְרִחֲמֶךָ וְשָׁב וְקִבֶּצְךָ מִכָּל הָעַמִּים אֲשֶׁר הֱפִיצְךָ יהוה (אדני) אלהיך שָׁמָּה: אִם יִהְיֶה נִדַּחֲךָ בִּקְצֵה הַשָּׁמָיִם מִשָּׁם יְקַבֶּצְךָ יהוה (אדני) אלהיך וּמִשָּׁם יִקָּחֲךָ: וֶהֱבִיאֲךָ יהוה (אדני) אלהיך אל הָאָרֶץ אֲשֶׁר יָרְשׁוּ אֲבֹתֶיךָ וִירִשְׁתָּהּ וְהֵיטִבְךָ וְהִרְבְּךָ מֵאֲבֹתֶיךָ: וְנֶאֱמַר עַל יְדֵי נְבִיאָךְ יהוה (אדני) יאהדונהי יהוה (אדני)
