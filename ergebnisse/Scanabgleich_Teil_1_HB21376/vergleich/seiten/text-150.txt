וַעַד
[בַּג' אוֹתִיּוֹת וְעֵד יְכַוֵּיֽן לְהוֹרִיד פ"ע דְּמוֹחִין דִּיסוֹד דב"ן לִיסוֹד דז"א. וִיכַוֵּיֽן לְחַבֵּר פ"ת דנח"י דמ"ה דז"א עִם ג"ע יְסוֹד דמ"ה דז"א וִיכַוֵּיֽן ד' אוֹתִיּוֹת יהוה וּלְמָשְׁכָם לנוק'. וב"פ עֶלְיוֹנִים דחו"מ דמ"ה דז"א שֶׁכְּנֶגֶד ב' אוֹתָיו' מ"ך דִּלְעוֹלָם שֶׁהֵם כְּמִסְפַּר מִיֽלּוּי דע"ב — הי ויו הי — יְכַוֵּיֽן לְהַעֲלוֹת פֶּרֶק ב' דְּנֶצַח דמ"ה וּפֶרֶק א' דִּיסוֹד דמ"ה דז"א. גַּם יְכַוֵּיֽן לְתַקֵּן מָל' דְּלֵאָה לְהַחֲזִירָהּ לַכְּתָרִים — הה הה הה הה. אַתָּה הוּא אֶחָד קֹוֽדֶם שֶׁבָּרָאתָ הָעוֹלָם וְאַתָּה הוּא אֶחָד לְאַחַר שֶׁבָּרָאתָ הָעוֹלָם אַתָּה הוּא אל בָּעוֹלָם הַזֶּה וְאַתָּה הוּא אל בָּעוֹלָם הַבָּא וְאַתָּה הוּא וּשְׁנוֹתֶיךָ לֹא יִתָּמּוּ. קֹדֶשׁ שִׁמְךָ בְּעוֹלָמֶךָ עַל עַם מַקְדִּישֵׁי שִׁמְךָ וּבִישׁוּעָתְךָ מַלְכֵּנוּ תָּרוּם וְתַגְבִּיהַּ קַרְנֵנוּ וְתוֹשִׁיעֵנוּ בְּקָרוֹב לְמַעַן שִׁמְךָ. בָּרוּךְ הַמְקַדֵּשׁ שְׁמוֹ בָּרַבִּים:
אַתָּה הוּא יהוה (אדני) האלהים [ר"ת אהיה] בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת בִּשְׁמֵי הַשָּׁמַיִם הָעֶלְיוֹנִים וְהַתַּחְתּוֹנִים אַתָּה הוּא רִאשׁוֹן וְאַתָּה הוּא אַחֲרוֹן וּמִבַּלְעָדֶיךָ אֵין אלהים קִבֵּץ נְפוּצוֹת קַוֶּיךָ מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ יַכִּירוּ וְיֵדְעוּ כָּל בָּאֵי עוֹלָם כִּי אַתָּה הוּא האלהים לְבַדְּךָ לְכֹל מַמְלְכוֹת
